        function loadStateFromStorage() {
            try {
                const saved = localStorage.getItem('fp3_trainer_state');

        function validatedState(raw) {
            if (!raw || typeof raw !== 'object' || Array.isArray(raw)) throw new Error('不正なバックアップ形式');
            const next = { ...appState };
            for (const [key, value] of Object.entries(raw)) {
                if (!['__proto__', 'prototype', 'constructor'].includes(key)) next[key] = value;
            }
            const maps = ['unitStatus', 'unitTestScores', 'chapterTestScores', 'dailyTimeLogs', 'customUnitDays', 'unitTestDetails', 'wrongQuestionRevisions'];
            for (const key of maps) {
                if (!next[key] || typeof next[key] !== 'object' || Array.isArray(next[key])) throw new Error('不正な履歴: ' + key);
                next[key] = { ...next[key] };
            }
            if (Object.hasOwn(raw, 'unitTestScores') && !Object.hasOwn(raw, 'unitTestDetails')) next.unitTestDetails = {};
            if (Object.hasOwn(raw, 'wrongQuestionIds') && !Object.hasOwn(raw, 'wrongQuestionRevisions')) next.wrongQuestionRevisions = {};
            if (Object.values(next.wrongQuestionRevisions).some(v => !Number.isInteger(v) || v < 1)) throw new Error('不正な復習問題の版');
            for (const detail of Object.values(next.unitTestDetails)) {
                if (!detail || !Number.isInteger(detail.total) || detail.total <= 0 || !Number.isInteger(detail.correct)
                    || detail.correct < 0 || detail.correct > detail.total || !['unit', 'chapter', 'practical'].includes(detail.mode)) throw new Error('不正な受検明細');
            }
            for (const key of ['unitTestScores', 'chapterTestScores']) {
                if (Object.values(next[key]).some(v => typeof v !== 'number' || !Number.isFinite(v) || v < 0 || v > 100)) throw new Error('不正な正答率');
            }
            if (Object.values(next.dailyTimeLogs).some(v => typeof v !== 'number' || !Number.isFinite(v) || v < 0)) throw new Error('不正な学習時間');
            if (Object.values(next.unitStatus).some(v => !['unstarted', 'in_progress', 'completed'].includes(v))) throw new Error('不正な読了状態');
            if (!Array.isArray(next.wrongQuestionIds) || next.wrongQuestionIds.some(v => typeof v !== 'string')) throw new Error('不正な復習履歴');
            if (!Array.isArray(next.testHistory)) throw new Error('不正なテスト履歴');
            if (typeof next.targetHours !== 'number' || !Number.isFinite(next.targetHours) || next.targetHours <= 0) throw new Error('不正な目標時間');
            if (!FP3Quiz.isDate(next.startDate) || (next.examDate && !FP3Quiz.isDate(next.examDate))) throw new Error('不正な日付');
            // 旧バックアップにタイマーがなければ、復元前の稼働状態を引き継がない。
            next.timerSession = raw.timerSession == null ? null : { ...raw.timerSession };
            if (raw.timerSession != null && (typeof raw.timerSession !== 'object' || Array.isArray(raw.timerSession)
                || (next.timerSession.id !== undefined && (typeof next.timerSession.id !== 'string' || !next.timerSession.id || next.timerSession.id.length > 200))
                || !Number.isSafeInteger(next.timerSession.elapsedMs) || next.timerSession.elapsedMs < 0
                || (next.timerSession.startedAt !== null && (!Number.isSafeInteger(next.timerSession.startedAt) || next.timerSession.startedAt < 0)))) {
                throw new Error('不正なタイマー履歴');
            }
            return next;
        }

        function renderDashboard() {
            // Days & Streak calculation
            const startDate = new Date(appState.startDate);
            const today = new Date();
            const diffTime = Math.abs(today - startDate);
            const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24)) + 1;
            const currentDayNumber = Math.min(Math.max(diffDays, 1), 10);

            document.getElementById('dash-day-badge').textContent = `DAY ${currentDayNumber} / 10`;
            document.getElementById('dash-days-left').innerHTML = `${11 - currentDayNumber} <span class="text-[10px] sm:text-xs font-normal">日</span>`;

            // Time calculations
            const todayStr = new Date().toISOString().split('T')[0];
            const todayMinutes = appState.dailyTimeLogs[todayStr] || 0;
            document.getElementById('dash-today-time').innerHTML = `${(todayMinutes / 60).toFixed(1)} <span class="text-[10px] sm:text-xs font-normal text-white">h / 5.0h</span>`;

            let totalMinutes = 0;
            Object.values(appState.dailyTimeLogs).forEach(m => totalMinutes += m);
            const totalHours = (totalMinutes / 60).toFixed(1);
            document.getElementById('dash-total-hours').textContent = `${totalHours} / ${appState.targetHours}.0h`;

            const timePercent = Math.min(Math.round((totalHours / appState.targetHours) * 100), 100);
            document.getElementById('dash-time-percent').textContent = `${timePercent}%`;
            document.getElementById('dash-time-remaining').textContent = `${Math.max(0, (appState.targetHours - totalHours).toFixed(1))}h`;

            // Text Progress
            let totalUnits = 0;
            let completedUnits = 0;
            let inProgressUnits = 0;
            let practicalCompleted = 0;
            let practicalTotal = 0;

            FP3_CURRICULUM.forEach(ch => {
                if (isOptionalChapter(ch)) return;
                const isPractical = ch.id === 'ch9';
                ch.units.forEach(u => {
                    totalUnits++;
                    if (isPractical) practicalTotal++;
                    const status = appState.unitStatus[u.id];
                    if (status === 'completed') {
                        completedUnits++;
                        if (isPractical) practicalCompleted++;
                    } else if (status === 'in_progress') {

            if (scores.length > 0) {

                document.getElementById('dash-understanding-percent').textContent = `${avgScore}%`;
                const statusBadge = document.getElementById('dash-understanding-status');

                    statusBadge.textContent = "順調";
                    statusBadge.className = "text-[10px] sm:text-xs font-bold px-1.5 py-0.5 rounded bg-emerald-100 text-emerald-700";
                } else if (avgScore >= 60) {
                    statusBadge.textContent = "合格ライン";
                    statusBadge.className = "text-[10px] sm:text-xs font-bold px-1.5 py-0.5 rounded bg-amber-100 text-amber-700";
                } else {
                    statusBadge.textContent = "要復習";
                    statusBadge.className = "text-[10px] sm:text-xs font-bold px-1.5 py-0.5 rounded bg-rose-100 text-rose-700";
                }
            } else {
                document.getElementById('dash-understanding-percent').textContent = '未受検';

            const container = document.getElementById('dash-weak-list');
            if (!container) return;

            const weakUnits = [];
            Object.entries(appState.unitTestScores).forEach(([uId, score]) => {
                if (score < 80 && requiredCurriculum().some(ch => ch.units.some(u => u.id === uId))) {
                    let uObj = null;
                    for (let ch of FP3_CURRICULUM) {
                        uObj = ch.units.find(u => u.id === uId);
                        if (uObj) break;
                    }
                    if (uObj) weakUnits.push({ unit: uObj, score });
                }
            });

            if (weakUnits.length === 0) {
                container.innerHTML = `<div class="text-xs text-slate-400 text-center py-4">現在の履歴に80%未満の単元はありません。未受検の単元の理解度は未確認です。</div>`;
                return;
            }

            weakUnits.sort((a, b) => a.score - b.score);
            weakUnits.slice(0, 3).forEach(item => {
                const el = document.createElement('div');
                el.className = 'p-2.5 rounded-xl border border-rose-100 bg-rose-50/50 flex justify-between items-center text-xs';

        // setIntervalの実行回数には依存せず、保存した開始時刻との差で計算する。
        // 開始・一時停止時に保存済みなので、電源断時のイベントやバックグラウンド実行は不要。
        function timerElapsedMs(now = Date.now()) {
            const session = appState.timerSession;
            if (!session) return 0;
            return session.elapsedMs + (session.startedAt === null ? 0 : Math.max(0, now - session.startedAt));
        }

        function restoreTimer() {
            clearInterval(timerInterval);
            timerInterval = null;
            isTimerRunning = !!appState.timerSession && appState.timerSession.startedAt !== null;
            if (isTimerRunning) timerInterval = setInterval(updateTimerDisplay, 1000);
            updateTimerDisplay();
            const active = appState.timerSession !== null;
            document.getElementById('timer-start-btn').innerHTML = isTimerRunning
                ? '<i class="fa-solid fa-pause mr-1"></i> 一時停止'
                : '<i class="fa-solid fa-play mr-1"></i> ' + (active ? '再開' : '開始');
            const stopButton = document.getElementById('timer-stop-btn');
            stopButton.disabled = !active;
            stopButton.className = (active ? 'bg-rose-500 hover:bg-rose-600 text-white' : 'bg-slate-200 text-slate-400 cursor-not-allowed')
                + ' font-semibold px-2.5 py-1.5 sm:px-4 sm:py-2.5 rounded-lg text-xs sm:text-sm flex items-center';
        }

        function toggleTimer() {
            const now = Date.now();
            const id = appState.timerSession?.id || crypto.randomUUID();
            appState.timerSession = isTimerRunning
                ? { id, elapsedMs: timerElapsedMs(now), startedAt: null }
                : { id, elapsedMs: timerElapsedMs(now), startedAt: now };
            saveStateToStorage();
            restoreTimer();
        }

        function stopTimer() {
            if (!appState.timerSession) return;
            const timeEventId = 'timer:' + (appState.timerSession.id || `legacy-${appState.timerSession.startedAt}-${appState.timerSession.elapsedMs}`);
            const minutesLearned = Math.round(timerElapsedMs() / 60000);
            if (minutesLearned > 0) {
                const today = new Date();
                const todayStr = `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, '0')}-${String(today.getDate()).padStart(2, '0')}`;
                appState.dailyTimeLogs[todayStr] = (appState.dailyTimeLogs[todayStr] || 0) + minutesLearned;
            }
            // 加算とセッションの解除を同じ保存に含め、再読込での二重記録を防ぐ。
            appState.timerSession = null;
            saveStateToStorage({ timeEventId });
            restoreTimer();
            renderAllViews();
            updateCharts();
            if (minutesLearned > 0) alert(`${minutesLearned}分間の学習を記録しました！お疲れ様でした。`);
        }

        function updateTimerDisplay() {
            timerSeconds = Math.floor(timerElapsedMs() / 1000);
            const hrs = String(Math.floor(timerSeconds / 3600)).padStart(2, '0');
            const mins = String(Math.floor((timerSeconds % 3600) / 60)).padStart(2, '0');
            const secs = String(timerSeconds % 60).padStart(2, '0');
            document.getElementById('timer-display').textContent = `${hrs}:${mins}:${secs}`;
        }

        document.addEventListener('visibilitychange', () => {
            if (!document.hidden) updateTimerDisplay();
        });
        window.addEventListener('pageshow', updateTimerDisplay);

        function openManualTimeModal() {
            document.getElementById('modal-manual-time').classList.remove('hidden');
        }

        function closeManualTimeModal() {
            document.getElementById('modal-manual-time').classList.add('hidden');
        }

        function saveManualTime() {
            const date = document.getElementById('manual-date').value;
            const mins = parseInt(document.getElementById('manual-minutes').value);
            if (!date || isNaN(mins) || mins <= 0) {
                alert('正しい日付と時間を分単位で入力してください。');
                return;
            }

            appState.dailyTimeLogs[date] = (appState.dailyTimeLogs[date] || 0) + mins;
            saveStateToStorage();
            closeManualTimeModal();
            renderAllViews();
            updateCharts();
            alert('学習時間を記録しました。');
        }

        function availableQuestions() {
            return FP3Quiz.available(QUESTION_DATABASE, FP3_REFERENCE_POLICY, appState.examDate);
        }

        function reviewQuestions() {
            return availableQuestions().filter(q => appState.wrongQuestionIds.includes(q.id)
                && (!q.bankEdition || appState.wrongQuestionRevisions[q.id] === q.revision));
        }

        function scoreHistoryLabel(unitId) {
            const detail = appState.unitTestDetails[unitId];
            if (!detail) return appState.unitTestScores[unitId] !== undefined ? '（旧履歴）' : '';
            const revised = QUESTION_DATABASE.some(q => q.unitId === unitId && q.bankEdition === FP3_BANK_EDITION);
            const old = revised && detail.bankEdition !== FP3_BANK_EDITION ? '・問題差替前の履歴' : '';
            const mode = { unit: '単元別', chapter: '章別抽出', practical: '実技20問抽出' }[detail.mode] || detail.mode;
            return `（${detail.total}問・${mode}${old}）`;
        }

        function unitInventory(unitId) {
            const rows = QUESTION_DATABASE.filter(q => q.unitId === unitId);
            const eligible = availableQuestions().filter(q => q.unitId === unitId);
            const unverified = rows.filter(q => q.verificationStatus !== 'verified').length;
            const label = rows.length === 0 ? '問題未収録' : rows.length < 5 ? '問題不足' : '収録済み';
            const shortage = eligible.length < 5 ? '・出題可能数が不足' : '';
            return { total: rows.length, available: eligible.length, unverified,
                text: `${label} ${rows.length}問 / 出題可能 ${eligible.length}問 / 未検証 ${unverified}問${shortage}` };
        }

        function safeText(value) {
            return String(value ?? '').replace(/[&<>"']/g, ch => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[ch]));
        }

        function isOptionalChapter(ch) { return ch.id === 'ch7' || ch.id === 'ch8'; }
        function requiredCurriculum() { return FP3_CURRICULUM.filter(ch => !isOptionalChapter(ch)); }
        function displayChapters() {
            return [...requiredCurriculum(), ...FP3_CURRICULUM.filter(isOptionalChapter)];
        }

        function renderQuestionCoverage() {
            const area = document.getElementById('question-coverage');
            const policy = FP3Quiz.referenceForDate(FP3_REFERENCE_POLICY, appState.examDate);
            const eligible = availableQuestions();
            const academic = FP3_CURRICULUM.filter(ch => /^ch[1-6]$/.test(ch.id)).flatMap(ch => ch.units);
            const bank = QUESTION_DATABASE.filter(q => /^ch[1-6]$/.test(q.chapterId));
            const covered = academic.filter(u => bank.some(q => q.unitId === u.id)).length;
            const ready = academic.filter(u => eligible.filter(q => q.unitId === u.id).length >= 5).length;
            const practical = QUESTION_DATABASE.filter(q => q.chapterId === 'ch9');
            const unverified = QUESTION_DATABASE.filter(q => q.verificationStatus !== 'verified').length;
            const requiredReady = requiredCurriculum().every(ch => ch.units.every(u => eligible.filter(q => q.unitId === u.id).length >= 5));
            const missingOptional = FP3_CURRICULUM.filter(isOptionalChapter).flatMap(ch => ch.units).filter(u => !QUESTION_DATABASE.some(q => q.unitId === u.id)).length;
            area.innerHTML = `<strong>${requiredReady ? '学科6章・日本FP協会実技の問題を収録済み' : 'この受検日の出題準備は未完了です'}</strong>
                <p>学科 ${academic.length}単元中、収録あり ${covered}単元・5問以上出題可能 ${ready}単元。
                学科 ${bank.length}問、日本FP協会実技 ${practical.length}問、任意学習を含む全体 ${QUESTION_DATABASE.length}問（未検証 ${unverified}問）。</p>
                <p>受検予定日: ${safeText(appState.examDate || '未設定')} / 法令基準日: ${policy ? safeText(policy.lawsDate) : '未確認（通常テストは開始できません）'}。
                <a href="https://www.jafp.or.jp/exam/schedule/" target="_blank" rel="noopener noreferrer" class="underline">日本FP協会の試験日程</a></p>
                <p>きんざい実技は任意学習です。未収録 ${missingOptional}単元は「問題未収録」と表示します。未検証・法令基準日不一致の問題は出題しません。各問の正解・解説・出典・検証日を収録状況で確認できます。</p>`;
            area.innerHTML += '<p class="mt-2">2024〜2026年の日本FP協会公開問題（学科180問・実技60問）を分析し、問題を再構成しました。補強問題は公開分での出題確認と区別しています。差替前の成績・誤答履歴は残し、版が一致しない誤答は保留にしています。</p>';
            document.getElementById('setting-exam-date').value = appState.examDate || '';
        }

        function showTestSelector(mode) {
            const area = document.getElementById('test-content-area');
            activeTestSession = null;
            if (!area) return;
            if (mode === 'unit') {
                const optionsHtml = displayChapters().map(ch => `<optgroup label="${safeText(ch.title)}${isOptionalChapter(ch) ? '【任意】' : ''}">${ch.units.map(u => {
                    const c = unitInventory(u.id);
                    return `<option value="${u.id}" ${!c.available ? 'disabled' : ''}>${safeText(u.title)}（${safeText(c.text)}）</option>`;
                }).join('')}</optgroup>`).join('');
                area.innerHTML = `<h3 class="font-bold text-slate-800 text-lg mb-2">単元別確認テスト</h3>
                    <p class="text-xs text-slate-500 mb-4">検証済みで受検日に適合する問題のみ出題します。出題可能数0の単元は選択できません。</p>
                    <div class="flex flex-wrap gap-3"><select id="selected-unit-test" aria-label="確認テストの単元" class="p-2 border rounded-lg text-sm max-w-full">${optionsHtml}</select>
                    <button onclick="startSingleUnitTest(document.getElementById('selected-unit-test').value)" class="bg-brand-magenta text-white px-4 py-2 rounded-lg text-sm">開始</button></div>`;
            } else if (mode === 'chapter') {
                const optionsHtml = displayChapters().map(ch => {
                    const n = availableQuestions().filter(q => q.chapterId === ch.id).length;
                    return `<option value="${ch.id}" ${n < 20 ? 'disabled' : ''}>${safeText(ch.title)}${isOptionalChapter(ch) ? '【任意】' : ''}（出題可能${n}問${n < 20 ? '・問題不足' : 'から最大25問'}）</option>`;
                }).join('');
                area.innerHTML = `<h3 class="font-bold text-slate-800 text-lg mb-2">章別確認テスト</h3>
                    <p class="text-xs text-slate-500 mb-4">最大25問を単元間で偏らないよう抽出します。20問未満の章は開始できません。</p>
                    <div class="flex flex-wrap gap-3"><select id="selected-chapter-test" aria-label="章別確認テストの章" class="p-2 border rounded-lg text-sm max-w-full">${optionsHtml}</select>
                    <button onclick="startChapterTest(document.getElementById('selected-chapter-test').value)" class="bg-emerald-600 text-white px-4 py-2 rounded-lg text-sm">開始</button></div>`;
            }
        }

        function startSingleUnitTest(unitId) {
            const unit = FP3_CURRICULUM.flatMap(ch => ch.units).find(u => u.id === unitId);
            const questions = availableQuestions().filter(q => q.unitId === unitId);
            if (!unit || !questions.length) {
                alert('問題未収録、未検証、または適用法令基準日が未確認のため、この単元のテストは開始できません。');
                return;
            }
            runTestEngine(FP3Quiz.shuffled(questions).slice(0, 10), `単元テスト：${unit.title}${questions.length < 5 ? '（問題不足）' : ''}`, { mode: 'unit', unitId });
        }

        function startChapterTest(chapterId) {
            const ch = FP3_CURRICULUM.find(ch => ch.id === chapterId);
            if (!ch) return;
            const questions = FP3Quiz.chapterSample(availableQuestions(), chapterId, ch.units.map(u => u.id));
            if (questions.length < 20) {
                alert('この章は出題可能な問題が20問未満です。未収録・未検証・適用基準日を確認してください。');
                return;
            }
            runTestEngine(questions, `章別テスト：${ch.title}`, { mode: 'chapter', chapterId });
        }

        function startPracticalTest() {
            const questions = FP3Quiz.practicalSample(availableQuestions());
            if (questions.length !== 20) { alert('実技20問を構成する分野の問題が不足、未検証、または法令基準日が未確認です。'); return; }
            runTestEngine(questions, '日本FP協会実技・公開問題構成の演習20問', { mode: 'practical', chapterId: 'ch9' });
        }

        function startReviewMode() {
            const questions = reviewQuestions();
            if (!questions.length) {
                alert(appState.wrongQuestionIds.length ? '履歴に誤答はありますが、現在復習できる同じ版の検証済み問題がありません。差替前の問題・旧ID・未検証問題の履歴は保留として保持しています。新版は単元別テストで練習できます。' : '現在、復習リストに登録された誤答問題はありません。');
                return;
            }
            runTestEngine(FP3Quiz.shuffled(questions), '弱点克服特訓', { mode: 'review' });
        }

        function runTestEngine(questions, title, context = {}) {
            if (!Array.isArray(questions) || !questions.length) { alert('出題可能な問題がありません。'); return false; }
            // 呼出元だけでなくエンジン入口でも確認する。候補URLの保存だけでは出題しない。
            const ids = new Set(availableQuestions().map(q => q.id));
            if (new Set(questions.map(q => q.id)).size !== questions.length || questions.some(q => !ids.has(q.id) || !QUESTION_DATABASE.includes(q))) {
                alert('未検証または不整合のある問題を含むため開始できません。'); return false;
            }
            activeTestSession = { title, context, questions, currentIndex: 0, userAnswers: [], complete: false };
            switchTab('tests');
            renderCurrentQuestion();
            return true;
        }

        function renderCurrentQuestion() {
            const session = activeTestSession;
            if (!session || session.complete) return;
            const q = session.questions[session.currentIndex];
            if (!q) return;
            const area = document.getElementById('test-content-area');
            area.innerHTML = `<div class="max-w-xl mx-auto space-y-4">
                <div class="flex justify-between text-xs text-slate-500 border-b pb-2"><span>${safeText(session.title)} (${session.currentIndex + 1} / ${session.questions.length})</span><span>検証済み</span></div>
                <p class="text-xs text-slate-500">法令基準日: ${safeText(q.lawsDate)} / 受検予定日: ${safeText(appState.examDate)}</p>
                <h3 class="text-base font-bold leading-relaxed">Q. ${safeText(q.question)}</h3>
                ${materialsMarkup(q)}
                <div class="space-y-2.5">${q.options.map((opt, idx) => `<button data-answer="${idx}" onclick="submitAnswer(${idx})" class="w-full text-left p-3.5 border-2 rounded-xl hover:border-brand-magenta">${safeText(opt)}</button>`).join('')}</div></div>`;
            area.scrollIntoView({ block: 'start' });
        }

        function materialsMarkup(q) {
            if (!q.materials) return '';
            return `<div class="overflow-x-auto"><table class="w-full text-xs text-left border border-slate-200">
                <caption class="text-left text-slate-500 mb-1">${safeText(q.materials.caption)}</caption>
                <thead class="bg-slate-100"><tr>${q.materials.columns.map(c => `<th class="p-2 border">${safeText(c)}</th>`).join('')}</tr></thead>
                <tbody>${q.materials.rows.map(row => `<tr>${row.map(cell => `<td class="p-2 border">${safeText(cell)}</td>`).join('')}</tr>`).join('')}</tbody></table></div>`;
        }

        function showPastExamAnalysis() {
            const s = FP3_PAST_EXAM_SUMMARY;
            document.getElementById('past-exam-analysis').innerHTML = `<p>${safeText(s.sampleNote)}</p>
                <p class="my-2">学科180問・実技60問を分析。既存${s.replacedQuestions}問を差し替え、${s.addedQuestions}問を追加しました。公開分と同じ論点の演習${s.observedTopicExercises}問、範囲の補強${s.supplementExercises}問です。</p>
                <p class="text-xs text-slate-500">下の観測数は、それぞれ3公開セットのうち対応設問があるセット数です。非公開CBTの出題率や今後の出題確率ではありません。</p>
                <div class="my-3 space-y-1">${s.papers.map(p => `<p><a class="underline" href="${safeText(p.questionUrl)}" target="_blank" rel="noopener noreferrer">${safeText(p.label)}</a> / 原問の法令基準日 ${safeText(p.lawsDate)}</p>`).join('')}</div>
                ${s.units.filter(u => !u.optional).map(u => `<details class="border rounded-lg p-3 my-2"><summary class="cursor-pointer font-semibold">${safeText(u.unitId)} ${safeText(u.title)}（学科${u.observedAcademicSets}/3・実技${u.observedPracticalSets}/3公開セット）</summary>
                    <p class="mt-2">${safeText(u.notes)}</p><p>収録${u.questionCount}問 / 差替${u.replaced}問 / 追加${u.added}問</p>
                    ${u.observedQuestions.length ? `<ul class="list-disc pl-5 my-2">${u.observedQuestions.map(q => `<li><a class="underline" href="${safeText(q.questionUrl)}" target="_blank" rel="noopener noreferrer">${safeText(q.label)} 問${q.number}</a>：${safeText(q.topic)}</li>`).join('')}</ul>` : '<p>分析した公開分では同じ論点は未確認。試験範囲の補強として区別しています。</p>'}</details>`).join('')}
                <p class="text-xs mt-3"><a class="underline" href="${safeText(s.licenseUrl)}" target="_blank" rel="noopener noreferrer">日本FP協会の試験問題の利用条件</a>に従い、出典と修正・加工を表示しています。</p>`;
        }

        function sourceMarkup(q) {
            const primary = q.sourceUrls.map((url, i) => `<a class="underline break-all" href="${safeText(url)}" target="_blank" rel="noopener noreferrer">出典${i + 1}: ${safeText(url)}</a>`).join('<br>');
            if (!q.bankEdition) return primary;
            const past = (q.pastExamRefs || []).map(ref => `<a class="underline" href="${safeText(ref.questionUrl)}" target="_blank" rel="noopener noreferrer">${safeText(ref.label)}</a>（${ref.relationship === 'same_topic' ? '同じ論点' : '関連論点の補強'}・修正加工）`).join('<br>');
            return primary + `<div class="mt-2 p-2 bg-white rounded"><strong>過去問分析からの練習問題</strong><p>${safeText(q.alignmentNote)}</p>${past || '<p>公開3セットで同じ論点は未確認。試験範囲の補強問題です。</p>'}<p>原問の転載ではなく、条件・数値・選択肢を再構成しています。掲載年は公表年です。</p></div>`;
        }

        function submitAnswer(chosenIdx) {
            const session = activeTestSession;
            if (!session || session.complete || session.userAnswers.length !== session.currentIndex) return;
            const q = session.questions[session.currentIndex];
            if (!q || !Number.isInteger(chosenIdx) || chosenIdx < 0 || chosenIdx >= q.options.length) return;
            const isCorrect = chosenIdx === q.answer;
            session.userAnswers.push({ questionId: q.id, chosen: chosenIdx, isCorrect });
            if (!isCorrect && !appState.wrongQuestionIds.includes(q.id)) appState.wrongQuestionIds.push(q.id);
            else if (isCorrect) appState.wrongQuestionIds = appState.wrongQuestionIds.filter(id => id !== q.id);
            if (!isCorrect) appState.wrongQuestionRevisions[q.id] = q.revision || 1;
            else delete appState.wrongQuestionRevisions[q.id];
            // 中断・再読込でも誤答履歴を失わない。旧IDや未検証IDは消さない。
            saveStateToStorage();
            document.getElementById('test-content-area').innerHTML = `<div class="max-w-xl mx-auto space-y-4">
                <h3 class="text-3xl font-bold ${isCorrect ? 'text-emerald-600' : 'text-rose-600'}">${isCorrect ? '正解！' : '不正解'}</h3>
                <p class="font-bold">正解: ${safeText(q.options[q.answer])}</p>
                <h4 class="font-bold leading-relaxed">Q. ${safeText(q.question)}</h4>
                ${materialsMarkup(q)}
                <div class="bg-slate-50 p-4 rounded-xl text-sm leading-relaxed"><strong>解説:</strong><p>${safeText(q.explanation)}</p></div>
                <div class="text-xs text-slate-500 space-y-1">法令基準日: ${safeText(q.lawsDate)} / 検証日: ${safeText(q.verifiedAt)} / 検証状態: 検証済み<br>${sourceMarkup(q)}</div>
                <button onclick="nextQuestion()" class="w-full bg-brand-magenta text-white font-bold py-3 rounded-xl">${session.currentIndex + 1 < session.questions.length ? '次の問題へ' : '結果を見る'}</button></div>`;
        }

        function nextQuestion() {
            const session = activeTestSession;
            if (!session || session.complete || session.userAnswers.length !== session.currentIndex + 1) return;
            session.currentIndex++;
            if (session.currentIndex < session.questions.length) renderCurrentQuestion();
            else renderTestResults();
        }

        function renderTestResults() {
            const session = activeTestSession;
            if (!session || session.complete || session.currentIndex !== session.questions.length || session.userAnswers.length !== session.questions.length) return;
            const result = FP3Quiz.results(session.questions, session.userAnswers);
            session.complete = true;
            const completedAt = new Date().toISOString();
            // 復習は抽出された誤答だけなので、単元全体の理解度を上書きしない。
            if (session.context.mode !== 'review') {
                Object.entries(result.units).forEach(([id, count]) => {
                    appState.unitTestScores[id] = Math.round(count.correct / count.total * 100);
                    appState.unitTestDetails[id] = { ...count, mode: session.context.mode, completedAt, bankEdition: session.questions.find(q => q.unitId === id).bankEdition || null };
                });
            }
            if (session.context.mode === 'chapter') appState.chapterTestScores[session.context.chapterId] = result.percent;
            appState.testHistory.push({ syncId: crypto.randomUUID(), mode: session.context.mode, chapterId: session.context.chapterId || null,
                examDate: appState.examDate, lawsDate: session.questions[0].lawsDate,
                completedAt, questionIds: session.questions.map(q => q.id), questionRevisions: session.questions.map(q => q.revision || 1), bankEdition: session.questions[0].bankEdition || null, answers: [...session.userAnswers],
                correct: result.correct, total: result.total, percent: result.percent });
            saveStateToStorage();
            document.getElementById('test-content-area').innerHTML = `<div class="max-w-md mx-auto text-center space-y-4 py-4">
                <h3 class="font-bold text-xl">テスト完了！</h3><p class="text-5xl font-bold">${result.percent}%</p>
                <p>正解: ${result.correct} / ${result.total} 問</p>
                <p class="text-xs text-slate-500">${session.context.mode === 'review' ? '復習履歴を保存しました。単元の理解度は上書きしません。' : '出題した単元ごとの理解度を保存しました。章別テストでは抽出された問題の正答率です。'}</p>

        function updateCharts() {
            let totalUnits = 0, completedUnits = 0;
            requiredCurriculum().forEach(ch => {
                ch.units.forEach(u => {
                    totalUnits++;
                    if (appState.unitStatus[u.id] === 'completed') completedUnits++;
                });
            });
            const textPct = totalUnits > 0 ? Math.round((completedUnits / totalUnits) * 100) : 0;

            if (chartTextProgress) {
                chartTextProgress.data.datasets[0].data = [textPct, 100 - textPct];
                chartTextProgress.update();
            }

            const requiredIds = new Set(requiredCurriculum().flatMap(ch => ch.units.map(u => u.id)));
            const scores = Object.entries(appState.unitTestScores).filter(([id]) => requiredIds.has(id)).map(([, score]) => score);
            const avgUnd = scores.length > 0 ? Math.round(scores.reduce((a, b) => a + b, 0) / scores.length) : 0;
            if (chartUnderstanding) {
                chartUnderstanding.data.datasets[0].data = [avgUnd, 100 - avgUnd];
                chartUnderstanding.update();
            }

            let totalMinutes = 0;
            Object.values(appState.dailyTimeLogs).forEach(m => totalMinutes += m);
            const timePct = Math.min(Math.round(((totalMinutes / 60) / appState.targetHours) * 100), 100);
            if (chartTimeAchievement) {
                chartTimeAchievement.data.datasets[0].data = [timePct, 100 - timePct];
                chartTimeAchievement.update();
            }
            if (chartDailyTime) {
                const start = new Date(appState.startDate + 'T00:00:00Z');
                chartDailyTime.data.datasets[0].data = Array.from({ length: 10 }, (_, i) => {
                    const day = new Date(start); day.setUTCDate(day.getUTCDate() + i);
                    return (appState.dailyTimeLogs[day.toISOString().slice(0, 10)] || 0) / 60;
                });
                chartDailyTime.update();
            }
            if (chartChapterAccuracy) {
                chartChapterAccuracy.data.datasets[0].data = displayChapters().map(ch => {
                    if (appState.chapterTestScores[ch.id] !== undefined) return appState.chapterTestScores[ch.id];
                    const values = ch.units.map(u => appState.unitTestScores[u.id]).filter(v => v !== undefined);
                    return values.length ? Math.round(values.reduce((a, b) => a + b, 0) / values.length) : null;
                });
                chartChapterAccuracy.update();
            }
        }

        function saveSettings() {
            const start = document.getElementById('setting-start-date').value;
            const hours = parseFloat(document.getElementById('setting-target-hours').value);
            const examDate = document.getElementById('setting-exam-date').value;
            if (examDate && !FP3Quiz.isDate(examDate)) { alert('受検予定日を確認してください。'); return; }
            appState.examDate = examDate;
            activeTestSession = null;

            if (start) appState.startDate = start;
            if (hours && hours > 0) appState.targetHours = hours;

            saveStateToStorage();
            renderAllViews();
            showQuestionInventory();
            document.getElementById('test-content-area').textContent = '設定を変更しました。テストのモードを選び直してください。';
            alert('設定を保存しました。');
        }

        function exportData() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState, null, 2));
            const dlAnchor = document.createElement('a');
            dlAnchor.setAttribute("href", dataStr);
            dlAnchor.setAttribute("download", `fp3_study_backup_${new Date().toISOString().split('T')[0]}.json`);
            document.body.appendChild(dlAnchor);
            dlAnchor.click();
            dlAnchor.remove();
        }

        function importData(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const imported = JSON.parse(e.target.result);
                    if (imported && typeof imported === 'object' && !Array.isArray(imported)) {
                        const restored = validatedState(imported);
                        appState = restored;
                        initializeDefaultState();
                        restoreTimer();
                        activeTestSession = null;
                        saveStateToStorage();
                        renderAllViews();
                        showQuestionInventory();
                        updateCharts();
                        document.getElementById('setting-start-date').value = appState.startDate;
                        document.getElementById('setting-target-hours').value = appState.targetHours;
                        document.getElementById('test-content-area').textContent = 'バックアップを復元しました。モードを選び直してください。';
                        alert('データを正常に復元しました！');
                    } else { throw new Error('不正なバックアップ'); }
                } catch (err) {
                    alert('ファイルの読み込みに失敗しました。正しいJSONファイルを選択してください。');
                }
            };
            reader.readAsText(file);
        }

        function resetAllData() {
            if (confirm('本当にすべての学習履歴を初期化しますか？この操作は取り消せません。')) {
                appState = JSON.parse(JSON.stringify(defaultStateTemplate));
                appState.startDate = new Date().toISOString().slice(0, 10);
                initializeDefaultState();
                activeTestSession = null;
                saveStateToStorage({ reset: true });
                restoreTimer(); renderAllViews(); updateCharts();
                document.getElementById('test-content-area').textContent = '初期化しました。モードを選び直してください。';
            }
        }
